use seleniumbase_rs::sb_cdp::{Browser, LaunchOptions, SelectBy};

#[tokio::test]
async fn recorded_test() -> Result<(), Box<dyn std::error::Error>> {
    let browser = Browser::launch(LaunchOptions::builder().build()?).await?;
    let page = browser.default_page().await?;
    page.goto("https://example.com/a?b=1&c=\"d\"").await?;
    page.locator("#go").click().await?;
    page.locator("input[name=\"q\"]").type_text("hello \"world\"\\n\nsecond line").await?;
    page.locator("#color").select_option(SelectBy::Text("Green")).await?;
    page.locator("#agree").check().await?;
    page.locator("#agree").uncheck().await?;
    page.locator("//div[@id=\"row\"]").double_click().await?;
    page.locator("[data-testid=\"menu\"]").hover().await?;
    page.locator("link=Sign in").click().await?;
    page.locator("#q").type_text("tab\there \u{202e} bidi \0 nul \u{1b} esc").await?;
    browser.close().await?;
    Ok(())
}
